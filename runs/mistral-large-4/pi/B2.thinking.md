store.ts addition:

export function resolveIncident(homeId: string, incidentId: string): Incident | undefined {
  const incident = incidents.find((i) => i.homeId === homeId && i.id === incidentId);
  if (!incident) return undefined;
  incident.status = "resolved";
  incident.endedAt = new Date().toISOString();
  return incident;
}
```

const resolveParams = z.object({
  homeId: z.string().min(1),

/** POST /api/homes/:homeId/incidents/:incidentId/resolve — mark an incident resolved. */
incidentsRouter.post("/:incidentId/resolve", (req, res) => {
  const params = resolveParams.safeParse(req.params);
  if (!params.success) {
    res.status(400).json({ error: "invalid_params", detail: params.error.issues });
    return;
  }

  const { homeId, incidentId } = params.data;
  if (!findHome(homeId)) {
    res.status(404).json({ error: "home_not_found", homeId });
    return;
  }

  const incident = resolveIncident(homeId, incidentId);
  if (!incident) {
    res.status(404).json({ error: "incident_not_found", incidentId });
    return;
  }

  res.json({ incident });
});
```
