```typescript
export type Contact = {
  email: string;
  unsubscribed?: boolean;
  hardBounced?: boolean;
};

export function contactsToSuppress(contacts: Contact[]): string[] {
  const emails = new Set<string>();

  for (const contact of contacts) {
    if (!contact.unsubscribed && !contact.hardBounced) continue;

    const email = contact.email.trim().toLowerCase();
    if (email.length > 0) {
      emails.add(email);
    }
  }

  return [...emails].sort();
}
```
