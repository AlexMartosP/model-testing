```ts
export type HourCell = {
  /** Start of the hour this cell covers, local time. */
  start: Date;
  severity: HourSeverity;
};

const HOUR_MS = 3_600_000;

/** The last 24 whole hours up to `now`, oldest first, each tagged with the worst incident that started in it. */
export function last24HourCells(
  incidents: { kind: IncidentKind; startedAt: string }[],
  now: Date = new Date(),
): HourCell[] {
  const currentHourStart = Math.floor(now.getTime() / HOUR_MS) * HOUR_MS;
  const cells: HourCell[] = [];
  for (let hoursAgo = 23; hoursAgo >= 0; hoursAgo--) {
    cells.push({ start: new Date(currentHourStart - hoursAgo * HOUR_MS), severity: "none" });
  }
  for (const incident of incidents) {
    const started = new Date(incident.startedAt).getTime();
    if (Number.isNaN(started)) continue;
    const hoursAgo = Math.floor((currentHourStart - started) / HOUR_MS);
    if (hoursAgo < 0 || hoursAgo > 23) continue;
    const cell = cells[23 - hoursAgo];
    if (severityRank[incident.kind] > severityRank[cell.severity]) {
      cell.severity = incident.kind;
    }
  }
  return cells;
}
```

```ts
export function summarizeHours(cells: HourCell[]): string {
  const counts: Record<HourSeverity, number> = { noise: 0, occupancy: 0, offline: 0, none: 0 };
  for (const cell of cells) counts[cell.severity]++;
  const parts: string[] = [];
  if (counts.noise) parts.push(`${counts.noise} noisy ${counts.noise === 1 ? "hour" : "hours"}`);
  if (counts.occupancy) parts.push(`${counts.occupancy} occupancy ${...}`);
  if (counts.offline) parts.push(`${counts.offline} device-offline ${...}`);
  parts.push(`${counts.none} quiet`);
  return `Last 24 hours: ${parts.join(", ")}.`;
}
```
