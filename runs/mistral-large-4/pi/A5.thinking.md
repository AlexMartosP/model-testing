```ts
export function totalDilution(steps: Step[]): number {
  let factor = 1;
  for (const [sample, diluent] of steps) {
    if (sample <= 0 || diluent <= 0) throw new RangeError(...);
    factor *= (sample + diluent) / sample;
  }
  return factor;
}
```
